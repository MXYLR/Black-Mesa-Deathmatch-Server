// tau_mp.games.txt — gamedata for tau_mp.smx
//
// 让单人战役的 tau cannon 拥有「多人模式」行为所需的两个代码位置
// （Black Mesa server.dll，md5 68278cc0c3a1e23d96c2b9d722ae7670。
//  单人战役与死亡服用的是同一份 server.dll，两边都复核过签名唯一命中）。
//
// BM 的 tau 用两处 IsMultiplayer 分支区分单人/多人
// （[g_pGameRules]->vtable+0x88，vtable 字节偏移 0x88 = 索引 34）：
//
//  1. CWeapon_Tau::FireBeam @ 0x1048C8D8
//         0F 84 DE 00 00 00    je 0x1048C9BC   ← 单人路径
//     单人路径（0x1048C9BC）算完击退的 X/Y 分量后，在 0x1048CA18 执行
//         mov dword ptr [ebp-0x18], 0
//     把击退向量的 **Z 分量清零** → 单人没有向上推力，即没有高斯跳。
//     多人路径（紧跟 je 的 0x1048C8DE）三轴都算，含 0x1048C955 读
//     owner 速度的 Z 分量、0x1048C9B5 写回 [ebp-0x18] → 有高斯跳。
//     两条路都在 0x1048CA3B 合流，用同一个 ConVar 缩放（X/Y 也同样缩放，
//     所以单人的水平击退是有的，缺的只有垂直分量）。
//
//  2. CWeapon_Tau::ChargeFire @ 0x1048D840
//         74 37                je 0x1048D879   ← 单人路径：副攻冷却 = curtime + 开火动画时长
//     紧随其后的 0x1048D842 是多人路径：SendWeaponAnim(0xB5) + 副攻冷却 = curtime（立即可再射）
//
// 插件用 GameConfGetAddress 取到这两条 je 指令的运行时地址，校验首字节后在内存里
// 把它们抹成 NOP（只改内存，不动磁盘文件、不动 cfg），于是引擎在这两处永远走多人分支。
// 行为 = 引擎自己的多人代码路径，不是重写的物理/冷却近似值。
//
// ---------------------------------------------------------------------------
// 签名必须「抗重定位」—— 这是 v1.1.0 高斯跳没生效的原因，务必记住：
//
// SourceMod 的 GameConfGetAddress 是在**进程内存**里扫描已加载的模块，而绝对地址类
// 指令（mov ecx,[abs] / cmp reg,imm32 / movss xmm,[abs]）的 32 位字段在加载时会被
// 基址重定位改写。只在磁盘文件上验证唯一命中是不够的 —— 签名一旦跨进这种字段，
// 磁盘上命中、运行时必然失配（日志表现为 "cannot find <key>"）。
//
// FireBeam 那条 je 之后紧跟 `cmp ecx, 0x110FA098`，其 imm32 在 0x1048C8E0 有重定位项；
// 所以整条 imm32 用 \x2A 通配，只保留与重定位无关的字节（寄存器/短立即数/相对位移）。
//
// 验证手段：smx_analysis/sig_check.py —— 它把 PE 的 base relocation 表施加到文件字节上，
// 模拟任意 ASLR 基址后再扫描，结果与运行时一致（已用它复现了 v1.1.0 的失配与本次的命中）。
//   python sig_check.py <server.dll> "<pattern>"
// 要求：磁盘镜像与加载镜像都只有 1 处命中，且地址就是上面注释的 je。
// ---------------------------------------------------------------------------

"Games"
{
	"#default"
	{
		"Addresses"
		{
			"TauFireBeamMPBranch"
			{
				"windows"
				{
					"signature"	"TauFireBeamMPBranch"
				}
			}
			"TauChargeFireMPBranch"
			{
				"windows"
				{
					"signature"	"TauChargeFireMPBranch"
				}
			}
		}

		"Signatures"
		{
			// je rel32 @ 0x1048C8D8
			//   0F 84 rel32                 je 0x1048C9BC        <- 通配跳转位移
			//   81 F9 imm32                 cmp ecx, 0x110FA098  <- imm32 有重定位，整体通配
			//   75 16                       jne 0x1048C8FC
			//   F3 0F 10 05 imm32           movss xmm0,[0x110FA0C4]  <- 同样通配
			//   F3 0F 11 45 FC              movss [ebp-4], xmm0
			//   81 75 FC imm32              xor   [ebp-4], 0x110FA098 <- 同样通配
			//   EB 0A                       jmp   0x1048C906
			// 尾部这段（movss/xor/jmp）完全不依赖重定位，用来把锚点钉死，避免短签名误命中。
			"TauFireBeamMPBranch"
			{
				"library"	"server"
				"windows"	"\x0F\x84\x2A\x2A\x2A\x2A\x81\xF9\x2A\x2A\x2A\x2A\x75\x16\xF3\x0F\x10\x05\x2A\x2A\x2A\x2A\xF3\x0F\x11\x45\xFC\x81\x75\xFC\x2A\x2A\x2A\x2A\xEB\x0A"
			}
			// je rel8 @ 0x1048D840；全段与重定位无关（寄存器/短立即数/相对位移），无需通配其余字节。
			"TauChargeFireMPBranch"
			{
				"library"	"server"
				"windows"	"\x74\x2A\x8B\x06\x8B\xCE\x68\xB5\x00\x00\x00\xFF\x90\x00\x04\x00\x00"
			}
		}
	}
}
