///////////////////////////////////////////////////////////////////////////////////////////////
//
// This special set of utilities are built for use in areas that have many conveyor belts, and as such will play everywhere.
// It will play in stereo, with no directional rotation.
//
// in addition to the local conveyor sounds placed in the map, 
// these utilities can act as "sweeteners" to pad out the void in between sound placements. or they can outright replace them in times of sound channel management.
//
// In situations where the player falls aways from the main playing area of conveyors, to a lower section of the same room per say,
//
// Volume attenuation of this conveyorbelt utility would be expected to occur, so to match the local sound placements as much as possible.
//
// To fake this,
//
// We set up a alternate version of the utility soundscape, where we then add the command "ambientpositionoverride", 
// setting a position via info_target in the vmf, above the plane at which the soundscape changeover would occur.
// 
// 
//
//////////////////////////////////////////////////////////////////////////////////////////////

// use this to play all over
// add ambientpositionoverride plus and info_target in vmf to fake attenuation when dropping to lower level areas of same room.

"c2a4.util_conveyorbelts_stereo" 
{
	"playlooping"
	{
		"volume"		"0.7"
		"pitch"			"100"
		"wave"			"bms_ambience\machine_ambience\m_ambience_conveyor1stereo.wav"
		"soundlevel"		"SNDLVL_85dB"
	}
